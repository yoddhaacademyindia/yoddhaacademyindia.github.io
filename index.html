function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);
    var type = data.type || "";

    // 1. TRAINER SYNC & IN-PLACE UPDATE (13 POINTS)
    if (type.indexOf("TRAINER") !== -1) {
      var sheet = getOrCreateSheet(ss, "Trainers", [
        "Timestamp", "Trainer ID", "Full Name", "Position", "Status",
        "Calling Mobile", "WhatsApp Mobile", "Official Email", "State",
        "District", "Akhada / Branch", "Full Address", "Login PIN / Password"
      ]);
      var trnId = data.trainerId || data.id || "";
      var row = [
        new Date(), trnId, data.fullName || data.name, data.position, data.status,
        data.mobile || data.phone, data.whatsapp, data.email, data.state,
        data.district, data.branch, data.address, data.password || data.pin
      ];
      var updated = updateRowIfExists(sheet, 2, trnId, row);
      if (!updated) sheet.appendRow(row);
      return ContentService.createTextOutput(JSON.stringify({ status: "success", updated: updated })).setMimeType(ContentService.MimeType.JSON);
    }

    // 2. STUDENT SYNC & IN-PLACE UPDATE (32 POINTS)
    if (type.indexOf("STUDENT") !== -1) {
      var sheet = getOrCreateSheet(ss, "Students", [
        "Timestamp", "Student ID", "Student Name", "Date of Birth", "Age",
        "Requested Level", "School Name", "Standard", "School Contact", "School Address",
        "Parent Name", "Parent Mobile", "City / Village", "District", "State",
        "Admission Date", "Trainer ID", "Trainer Name", "Status", "Fee Status",
        "Fee Amount", "UPI UTR Number"
      ]);
      var stuId = data.studentId || data.id || "";
      var row = [
        new Date(), stuId, data.studentName || data.name, data.dob, data.age,
        data.requestedLevel, data.schoolName, data.standard, data.schoolContact, data.schoolAddress,
        data.parentName, data.parentMobile, data.city, data.district, data.state,
        data.admissionDate, data.trainerId, data.trainerName, data.status, data.feeStatus,
        data.feeAmount, data.utrNumber
      ];
      var updated = updateRowIfExists(sheet, 2, stuId, row);
      if (!updated) sheet.appendRow(row);
      return ContentService.createTextOutput(JSON.stringify({ status: "success", updated: updated })).setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function getOrCreateSheet(ss, name, headers) {
  var s = ss.getSheetByName(name);
  if (!s) {
    s = ss.insertSheet(name);
    s.appendRow(headers);
    var r = s.getRange(1, 1, 1, headers.length);
    r.setBackground("#ea580c");
    r.setFontColor("#ffffff");
    r.setFontWeight("bold");
    s.setFrozenRows(1);
  }
  return s;
}

function updateRowIfExists(sheet, idCol, idVal, rowData) {
  if (!idVal) return false;
  var lastRow = sheet.getLastRow();
  if (lastRow <= 1) return false;
  var values = sheet.getRange(2, idCol, lastRow - 1, 1).getValues();
  for (var i = 0; i < values.length; i++) {
    if (String(values[i][0]).trim().toUpperCase() === String(idVal).trim().toUpperCase()) {
      sheet.getRange(i + 2, 1, 1, rowData.length).setValues([rowData]);
      return true;
    }
  }
  return false;
}
